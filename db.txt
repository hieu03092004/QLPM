SELECT pg_get_functiondef(17198);
CREATE OR REPLACE FUNCTION public.get_lich_thuc_hanh_sinh_vien(p_sinh_vien_id uuid, p_ngay date)
 RETURNS TABLE(ten_hoc_phan text, ten_giang_vien text, phong_thuc_hanh text, thoi_gian text, so_luong_sinh_vien integer)
 LANGUAGE plpgsql
AS $function$
BEGIN
  RETURN QUERY
  WITH thoi_gian_ca_hoc AS (
    SELECT 
      cht.ma_ca_thu,
      CONCAT(ca.gio_bat_dau, '-', ca.gio_ket_thuc) AS thoi_gian_hoc
    FROM "CaHoc_ThuTrongTuan" cht
    JOIN "CaHoc" ca 
      ON ca.ma_ca_hoc = cht.ma_ca
  )
  , sv_filter AS (
    -- lấy từng nhóm + ca mà sinh viên này tham gia vào ngày p_ngay
    SELECT 
      nth.ma_nhom_thuc_hanh,
      nth.phong_thuc_hanh,
      hp.ten_hoc_phan,
      tk.ho_va_ten        AS ten_giang_vien,
      tgch.thoi_gian_hoc,
      sv.ma_so_tai_khoan_sinh_vien
    FROM "SinhVienNhomThucHanh" sv
    JOIN "NhomThucHanh" nth 
      ON sv.ma_nhom_thuc_hanh     = nth.ma_nhom_thuc_hanh
    JOIN "PhieuDangKy" pdk 
      ON pdk.ma_nhom_thuc_hanh    = nth.ma_nhom_thuc_hanh
    JOIN "ChiTietDangKy" ctdk 
      ON ctdk.ma_phieu            = pdk.ma_phieu
     AND ctdk.ngay_hoc            = p_ngay
    JOIN thoi_gian_ca_hoc tgch 
      ON ctdk.ma_ca_thu           = tgch.ma_ca_thu
    JOIN "LopHocPhan" lhp 
      ON nth.ma_lop_hoc_phan      = lhp.ma_lop_hoc_phan
    JOIN "HocPhan" hp 
      ON lhp.ma_hoc_phan          = hp.ma_hoc_phan
    JOIN "TaiKhoan" tk 
      ON lhp.ma_tai_khoan_can_bo_giang_day = tk.ma_so_tai_khoan
    WHERE sv.ma_so_tai_khoan_sinh_vien = p_sinh_vien_id
  )
  , group_counts AS (
    -- đếm tổng số thành viên của mỗi nhóm trong sv_filter
    SELECT 
      ma_nhom_thuc_hanh,
      COUNT(*)::INT AS total_members
    FROM "SinhVienNhomThucHanh"
    WHERE ma_nhom_thuc_hanh IN (
      SELECT DISTINCT ma_nhom_thuc_hanh FROM sv_filter
    )
    GROUP BY ma_nhom_thuc_hanh
  )
  SELECT
    sf.ten_hoc_phan       ::TEXT   AS ten_hoc_phan,
    sf.ten_giang_vien     ::TEXT   AS ten_giang_vien,
    sf.phong_thuc_hanh    ::TEXT   AS phong_thuc_hanh,
    sf.thoi_gian_hoc      ::TEXT   AS thoi_gian,
    gc.total_members             AS so_luong_sinh_vien
  FROM sv_filter sf
  JOIN group_counts gc
    ON gc.ma_nhom_thuc_hanh = sf.ma_nhom_thuc_hanh;
END;
$function$
